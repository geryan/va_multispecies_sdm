# Pairwise agreement in the *location* of novel cells (Jaccard-style overlap)
overlap <- function(a, b) {
  both <- a & b
  either <- a | b
  sum(both, na.rm = TRUE) / sum(either, na.rm = TRUE)
}

# Correlation of the negative MESS values where both species are novel
neg_cor <- function(a, b) {
  va <- values(a)[, 1]; vb <- values(b)[, 1]
  ok <- !is.na(va) & !is.na(vb)
  if (sum(ok) < 2) return(NA_real_)
  cor(va[ok], vb[ok])
}

MESS surfaces and iterative novel-environment point search

For each of the three target species, iteratively generate a MESS surface (starting from the iteration 0 surface above), identify the cell with the lowest similarity score, record its centroid, add it to the reference set, and repeat until 20 points have been recorded.

n_iter <- 20
mess_results <- list()

for (sp in target_species) {

  # Start from the species observation points
  ref_vals <- ref_vals_0[[sp]]

  recorded_pts <- data.frame(longitude = numeric(0), latitude = numeric(0))
  # Surfaces indexed 1..(n_iter + 1) for iterations 0..n_iter; iteration 0 is
  # the MESS before any search points have been added
  mess_surfaces <- vector("list", n_iter + 1)

  for (i in 0:n_iter) {

    # Reuse the pre-computed iteration 0 surface; otherwise recompute
    m <- if (i == 0) mess_iter0[[sp]] else mess(cov_rast_ken_sub, ref_vals)
    mess_surfaces[[i + 1]] <- m

    # Final surface computed; no further points to add
    if (i == n_iter) break

    # Identify cell with lowest MESS value and get its centroid
    min_cell <- which.min(values(m$mess))
    centroid  <- xyFromCell(m$mess, min_cell)
    lon <- centroid[1, "x"]
    lat <- centroid[1, "y"]

    recorded_pts <- rbind(
      recorded_pts,
      data.frame(longitude = lon, latitude = lat)
    )

    # Extract env values at new point and add to reference set
    new_pt   <- vect(data.frame(x = lon, y = lat), geom = c("x", "y"), crs = crs(cov_rast_ken_sub))
    new_vals <- terra::extract(cov_rast_ken_sub, new_pt, ID = FALSE)
    ref_vals <- rbind(ref_vals, new_vals)
  }

  mess_results[[sp]] <- list(
    mess_surfaces = mess_surfaces,
    recorded_points = recorded_pts
  )

  message("Done: ", sp)
}

Absolute sum of positive and negative cells per iteration

For each surface (from iteration 0), sum the positive cell values (similar, blue) and the absolute value of the negative cells (novel, red), then track how these totals change across iterations. The grey line (secondary axis) shows the MESS value of the point selected at each iteration – i.e. how novel the most novel cell is – so we can see what is being picked as the search continues.

cell_sums <- lapply(target_species, function(sp) {
  surfaces <- mess_results[[sp]]$mess_surfaces
  data.frame(
    species   = sp,
    iteration = seq_along(surfaces) - 1,
    positive  = sapply(surfaces, function(m) {
      v <- values(m$mess); sum(v[v > 0], na.rm = TRUE)
    }),
    negative  = sapply(surfaces, function(m) {
      v <- values(m$mess); sum(abs(v[v < 0]), na.rm = TRUE)
    })
  )
}) |>
  bind_rows() |>
  pivot_longer(c(positive, negative), names_to = "sign", values_to = "abs_sum")

# MESS value of the point selected at each iteration (the most novel cell).
# Points are selected at iterations 0..(n_iter - 1); the final surface has none.
selected_mess <- lapply(target_species, function(sp) {
  surfaces <- mess_results[[sp]]$mess_surfaces
  data.frame(
    species    = sp,
    iteration  = seq_len(n_iter) - 1,
    mess_value = sapply(surfaces[seq_len(n_iter)], function(m) {
      min(values(m$mess), na.rm = TRUE)
    })
  )
}) |>
  bind_rows()

Funestus-driven point selection

Here the iterative search selects points only from the funestus MESS surface, but each selected point is added to all three species’ reference sets. This shows how points chosen to fill funestus’s novel environment reshape the MESS of the other two species as well.

# Reference sets start from each species' observation points
ref_vals_fun <- ref_vals_0

recorded_pts_fun   <- data.frame(longitude = numeric(0), latitude = numeric(0))
selected_cells_fun <- integer(0)
mess_surfaces_fun  <- setNames(
  lapply(target_species, \(sp) vector("list", n_iter + 1)),
  target_species
)

for (i in 0:n_iter) {

  # MESS for every species with the current (shared) reference sets
  for (sp in target_species) {
    mess_surfaces_fun[[sp]][[i + 1]] <-
      if (i == 0) mess_iter0[[sp]] else mess(cov_rast_ken_sub, ref_vals_fun[[sp]])
  }

  if (i == n_iter) break

  # Select the most novel cell from the FUNESTUS surface only
  m_fun    <- mess_surfaces_fun[["funestus"]][[i + 1]]
  min_cell <- which.min(values(m_fun$mess))
  centroid <- xyFromCell(m_fun$mess, min_cell)
  lon <- centroid[1, "x"]
  lat <- centroid[1, "y"]

  recorded_pts_fun   <- rbind(recorded_pts_fun, data.frame(longitude = lon, latitude = lat))
  selected_cells_fun <- c(selected_cells_fun, min_cell)

  # Add the funestus-selected point to ALL species' reference sets
  new_pt   <- vect(data.frame(x = lon, y = lat), geom = c("x", "y"), crs = crs(cov_rast_ken_sub))
  new_vals <- terra::extract(cov_rast_ken_sub, new_pt, ID = FALSE)
  for (sp in target_species) ref_vals_fun[[sp]] <- rbind(ref_vals_fun[[sp]], new_vals)
}

# Package like mess_results; recorded points are shared (the funestus picks)
mess_results_fun <- setNames(
  lapply(target_species, \(sp) list(
    mess_surfaces   = mess_surfaces_fun[[sp]],
    recorded_points = recorded_pts_fun
  )),
  target_species
)

CV-weighted MESS point selection

Here the species-by-species iterative search is repeated, but instead of selecting the cell with the lowest MESS value, we select based on a weighted combination: weight = MESS × CV, where CV is the coefficient of variation in the covariates at each pixel (high CV = high covariate diversity, worth exploring).

n_iter <- 20
mess_results_cv <- list()

for (sp in target_species) {

  # Start from the species observation points
  ref_vals <- ref_vals_0[[sp]]

  recorded_pts_cv <- data.frame(longitude = numeric(0), latitude = numeric(0))
  mess_surfaces_cv <- vector("list", n_iter + 1)

  for (i in 0:n_iter) {

    # Reuse the pre-computed iteration 0 surface; otherwise recompute
    m <- if (i == 0) mess_iter0[[sp]] else mess(cov_rast_ken_sub, ref_vals)
    mess_surfaces_cv[[i + 1]] <- m

    # Final surface computed; no further points to add
    if (i == n_iter) break

    # Weight by CV: low MESS (novel) × high CV (diverse covariates)
    # Both are masked to Kenya; multiply MESS by CV to get the weighted surface
    weighted <- m$mess * cv[[sp]]

    # Select cell with lowest weighted value (most novel and diverse)
    min_cell <- which.min(values(weighted))
    centroid  <- xyFromCell(weighted, min_cell)
    lon <- centroid[1, "x"]
    lat <- centroid[1, "y"]

    recorded_pts_cv <- rbind(
      recorded_pts_cv,
      data.frame(longitude = lon, latitude = lat)
    )

    # Extract env values at new point and add to reference set
    new_pt   <- vect(data.frame(x = lon, y = lat), geom = c("x", "y"), crs = crs(cov_rast_ken_sub))
    new_vals <- terra::extract(cov_rast_ken_sub, new_pt, ID = FALSE)
    ref_vals <- rbind(ref_vals, new_vals)
  }

  mess_results_cv[[sp]] <- list(
    mess_surfaces = mess_surfaces_cv,
    recorded_points = recorded_pts_cv
  )

  message("Done (CV-weighted): ", sp)
}

Combined CV-weighted MESS point selection (all species)

Here the iteration selects a single location per iteration from across all three species by computing MESS × CV for each species and cell, then picking the cell (from any species) with the lowest weighted value. This matches the species-by-species selection direction: because MESS is strongly negative in novel areas, MESS × CV is most negative where a cell is both novel and high-CV, so the minimum identifies the most novel-and-diverse cell. This location is then added to all three species’ reference sets, so all species evolve together around shared points.

ref_vals_combined <- ref_vals_0
recorded_pts_combined <- data.frame(longitude = numeric(0), latitude = numeric(0))
mess_surfaces_combined <- setNames(
  lapply(target_species, \(sp) vector("list", n_iter + 1)),
  target_species
)

for (i in 0:n_iter) {

  # MESS for every species with the current (shared) reference sets
  for (sp in target_species) {
    mess_surfaces_combined[[sp]][[i + 1]] <-
      if (i == 0) mess_iter0[[sp]] else mess(cov_rast_ken_sub, ref_vals_combined[[sp]])
  }

  if (i == n_iter) break

  # For each species and cell, compute weight = MESS × CV
  # Then select the cell with the lowest weight across all species (consistent
  # with the species-by-species which.min selection)
  weights <- lapply(target_species, function(sp) {
    mess_surfaces_combined[[sp]][[i + 1]]$mess * cv[[sp]]
  })
  names(weights) <- target_species

  # Stack all weighted surfaces and find the global minimum
  all_weights <- lapply(weights, \(w) values(w))
  min_idx <- which.min(unlist(all_weights))

  # Identify which species contributed the min and recover the cell index
  cumsum_len <- cumsum(sapply(all_weights, length))
  sp_idx <- sum(min_idx > c(0, cumsum_len[-length(cumsum_len)]))
  sp_name <- target_species[sp_idx]
  cell_in_sp <- min_idx - c(0, cumsum_len[-length(cumsum_len)])[sp_idx]

  # Get the cell centroid
  centroid <- xyFromCell(weights[[sp_name]], cell_in_sp)
  lon <- centroid[1, "x"]
  lat <- centroid[1, "y"]

  recorded_pts_combined <- rbind(
    recorded_pts_combined,
    data.frame(longitude = lon, latitude = lat, source_species = sp_name)
  )

  # Add the point to ALL species' reference sets
  new_pt <- vect(data.frame(x = lon, y = lat), geom = c("x", "y"), crs = crs(cov_rast_ken_sub))
  new_vals <- terra::extract(cov_rast_ken_sub, new_pt, ID = FALSE)
  for (sp in target_species) {
    ref_vals_combined[[sp]] <- rbind(ref_vals_combined[[sp]], new_vals)
  }
}

MESS surface evolution by species (combined CV-weighted)

The same shared points (from the combined optimization) are overlaid on each species’ surface, showing how the globally-optimized picks affect each species.

clamp_absmax_combined <- max(abs(unlist(lapply(target_species, function(sp) {
  lapply(mess_results_combined[[sp]]$mess_surfaces, function(m) {
    quantile(values(m$mess), c(0.01, 0.99), na.rm = TRUE)
  })
}))))

mess_range_combined <- c(-clamp_absmax_combined, clamp_absmax_combined)

for (sp in target_species) {
  pts      <- mess_results_combined[[sp]]$recorded_points
  surfaces <- mess_results_combined[[sp]]$mess_surfaces

  layout(layout_mat, heights = c(1, 1, 1, 0.4))
  par(mar = c(1, 1, 2, 0.5), oma = c(0, 0, 3, 0))

  for (j in seq_len(9)) {
    iter <- j - 1
    plot(
      surfaces[[j]]$mess,
      main   = paste0("iter ", iter),
      col    = divergent_pal,
      range  = mess_range_combined,
      legend = FALSE,
      axes   = FALSE
    )
    lines(ken)

    if (iter >= 1) {
      pts_so_far <- pts[seq_len(iter), , drop = FALSE]
      pts_vect   <- vect(pts_so_far, geom = c("longitude", "latitude"), crs = crs(cov_rast_ken_sub))
      points(pts_vect, pch = 19, col = "goldenrod", cex = 2)
    }
  }

  par(mar = c(2.5, 3, 0.5, 3))
  legend_vals <- seq(mess_range_combined[1], mess_range_combined[2], length.out = length(divergent_pal))
  image(
    x = legend_vals, y = 1, z = matrix(legend_vals, ncol = 1),
    col = divergent_pal, axes = FALSE, xlab = "MESS (clamped to 1st/99th pctile)", ylab = ""
  )
  axis(1)

  mtext(paste0(sp, " (combined CV-weighted)"), outer = TRUE, cex = 1.4, font = 2)
}
